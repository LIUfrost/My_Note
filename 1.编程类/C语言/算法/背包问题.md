### 1.01背包问题
**问题描述**：有N件物品，每件物品有重量w[i]和价值v[i]，每种物品只有一件，求在容量为V的背包中能获得的最大价值。
```C
#include <stdio.h>
#include <string.h>
#define MAXN 1005
#define MAXV 1005

int dp[MAXV]; // dp[j]表示容量为j时的最大价值
int w[MAXN], v[MAXN];

// 01背包标准解法
void zeroOnePack(int n, int V) {
    memset(dp, 0, sizeof(dp));
    for (int i = 1; i <= n; i++) {
        // 逆序枚举容量，保证每个物品只选一次
        for (int j = V; j >= w[i]; j--) {
            if (dp[j] < dp[j - w[i]] + v[i]) {
                dp[j] = dp[j - w[i]] + v[i];
            }
        }
    }
    printf("%d\n", dp[V]);
}

int main() {
    int n, V;
    scanf("%d%d", &n, &V);
    for (int i = 1; i <= n; i++) {
        scanf("%d%d", &w[i], &v[i]);
    }
    zeroOnePack(n, V);
    return 0;
}
```
假设：
- V=10（容量10）
- 物品1：重量6，价值5
- 物品2：重量4，价值4
运行过程：
1. 处理物品1后：dp[6]=5, dp[7]=5, dp[8]=5, dp[9]=5, dp[10]=5
2. 处理物品2时逆序：
    - j=10: dp[10]=max(5, dp[6]+4)=max(5,5+4)=9
    - j=9: dp[9]=max(5, dp[5]+4)=5
    - j=8: dp[8]=max(5, dp[4]+4)=max(5,0+4)=5
    - j=7: dp[7]=max(5, dp[3]+4)=5
    - j=6: dp[6]=max(5, dp[2]+4)=5
    - j=5: dp[5]=max(0, dp[1]+4)=4
    - j=4: dp[4]=max(0, dp[0]+4)=4
### 2.完全背包问题
**问题描述**：每件物品可以无限次使用，求最大价值。
```C
// 完全背包标准解法
void completePack(int n, int V) {
    memset(dp, 0, sizeof(dp));
    for (int i = 1; i <= n; i++) {
        // 正序枚举容量，允许重复选择
        for (int j = w[i]; j <= V; j++) {
            if (dp[j] < dp[j - w[i]] + v[i]) {
                dp[j] = dp[j - w[i]] + v[i];
            }
        }
    }
    printf("%d\n", dp[V]);
}
```

### 3.多重背包问题
**问题描述**：每种物品有有限个c[i]件。
#### **方法一：二进制优化**
```C
#include <stdio.h>
#include <string.h>
#define MAXN 100005

int dp[MAXV];
int w[MAXN], v[MAXN];
int cnt = 0; // 拆分后的物品数量

// 二进制拆分
void binarySplit(int n) {
    cnt = 0;
    for (int i = 1; i <= n; i++) {
        int weight, value, num;
        scanf("%d%d%d", &weight, &value, &num);
        
        // 二进制拆分
        for (int k = 1; k <= num; k <<= 1) {
            cnt++;
            w[cnt] = weight * k;
            v[cnt] = value * k;
            num -= k;
        }
        if (num > 0) {
            cnt++;
            w[cnt] = weight * num;
            v[cnt] = value * num;
        }
    }
}

void multiplePack(int n, int V) {
    binarySplit(n);
    memset(dp, 0, sizeof(dp));
    
    // 对拆分后的物品做01背包
    for (int i = 1; i <= cnt; i++) {
        for (int j = V; j >= w[i]; j--) {
            if (dp[j] < dp[j - w[i]] + v[i]) {
                dp[j] = dp[j - w[i]] + v[i];
            }
        }
    }
    printf("%d\n", dp[V]);
}
```
#### **方法二：单调队列优化**
```c
#include <stdio.h>
#include <string.h>
#define MAXV 20005

int dp[MAXV], q[MAXV], pre[MAXV];

void multiplePackOptimized(int n, int V) {
    memset(dp, 0, sizeof(dp));
    
    for (int i = 1; i <= n; i++) {
        int weight, value, num;
        scanf("%d%d%d", &weight, &value, &num);
        
        // 复制当前dp状态
        memcpy(pre, dp, sizeof(dp));
        
        // 按余数分类
        for (int r = 0; r < weight; r++) {
            int head = 0, tail = -1;
            for (int j = r; j <= V; j += weight) {
                // 维护队列，保证队首最大
                while (head <= tail && q[head] < j - num * weight) {
                    head++;
                }
                // 维护单调递减队列
                while (head <= tail && pre[q[tail]] - (q[tail] - r) / weight * value <= pre[j] - (j - r) / weight * value) {
                    tail--;
                }
                q[++tail] = j;
                dp[j] = pre[q[head]] + (j - q[head]) / weight * value;
            }
        }
    }
    printf("%d\n", dp[V]);
}
```
### 4.混合背包问题
**问题描述**：三种背包的混合，有的物品只能选1件，有的可选无限，有的可选有限件
```C
void mixedPack(int n, int V) {
    memset(dp, 0, sizeof(dp));
    
    for (int i = 1; i <= n; i++) {
        int weight, value, type;
        scanf("%d%d%d", &weight, &value, &type);
        
        if (type == 0) { // 完全背包
            for (int j = weight; j <= V; j++) {
                if (dp[j] < dp[j - weight] + value) {
                    dp[j] = dp[j - weight] + value;
                }
            }
        } else { // 01背包或多重背包（用二进制优化）
            int num = type;
            if (type == -1) num = 1; // 01背包
            
            // 二进制拆分
            for (int k = 1; k <= num; k <<= 1) {
                int ww = weight * k;
                int vv = value * k;
                for (int j = V; j >= ww; j--) {
                    if (dp[j] < dp[j - ww] + vv) {
                        dp[j] = dp[j - ww] + vv;
                    }
                }
                num -= k;
            }
            if (num > 0) {
                int ww = weight * num;
                int vv = value * num;
                for (int j = V; j >= ww; j--) {
                    if (dp[j] < dp[j - ww] + vv) {
                        dp[j] = dp[j - ww] + vv;
                    }
                }
            }
        }
    }
    printf("%d\n", dp[V]);
}
```
### 5.二维背包问题
**问题描述**：每件物品有两个维度的限制（如重量和体积）
```c
#define MAXN 105
#define MAXV 105
#define MAXM 105

int dp[MAXV][MAXM]; // dp[j][k]表示容量j，体积k的最大价值
int w[MAXN], v[MAXN], c[MAXN]; // w重量，v体积，c价值

void twoDimensionPack(int n, int V, int M) {
    memset(dp, 0, sizeof(dp));
    
    for (int i = 1; i <= n; i++) {
        // 二维费用，两层逆序
        for (int j = V; j >= w[i]; j--) {
            for (int k = M; k >= c[i]; k--) {
                if (dp[j][k] < dp[j - w[i]][k - c[i]] + v[i]) {
                    dp[j][k] = dp[j - w[i]][k - c[i]] + v[i];
                }
            }
        }
    }
    printf("%d\n", dp[V][M]);
}
```
### 6.分组背包问题
**问题描述**：物品分为若干组，每组最多选一个物品。
```c
#define MAXN 105
#define MAXG 105

int dp[MAXV];
int w[MAXN][MAXN], v[MAXN][MAXN], num[MAXN]; // num[i]表示第i组的物品数量

void groupPack(int n, int V) {
    memset(dp, 0, sizeof(dp));
    
    for (int i = 1; i <= n; i++) { // 遍历每组
        // 逆序枚举容量，保证每组只选一个
        for (int j = V; j >= 0; j--) {
            // 遍历组内物品
            for (int k = 1; k <= num[i]; k++) {
                if (j >= w[i][k]) {
                    if (dp[j] < dp[j - w[i][k]] + v[i][k]) {
                        dp[j] = dp[j - w[i][k]] + v[i][k];
                    }
                }
            }
        }
    }
    printf("%d\n", dp[V]);
}
```

### 7.有依赖的背包问题
**问题描述**：物品之间有依赖关系，选主件才能选附件。
```c
#include <stdio.h>
#include <string.h>
#define MAXN 65
#define MAXV 32005

int dp[MAXV];
int mainW[MAXN], mainV[MAXN]; // 主件重量和价值
int attachW[MAXN][3], attachV[MAXN][3]; // 附件
int attachNum[MAXN]; // 附件数量

void dependentPack(int n, int V) {
    memset(dp, 0, sizeof(dp));
    
    for (int i = 1; i <= n; i++) {
        if (mainW[i] == 0) continue; // 不是主件跳过
        
        // 逆序枚举容量
        for (int j = V; j >= mainW[i]; j--) {
            // 不选附件
            int maxVal = dp[j - mainW[i]] + mainV[i];
            
            // 选第一个附件
            if (attachNum[i] >= 1 && j >= mainW[i] + attachW[i][1]) {
                int val = dp[j - mainW[i] - attachW[i][1]] + mainV[i] + attachV[i][1];
                if (val > maxVal) maxVal = val;
            }
            
            // 选第二个附件
            if (attachNum[i] >= 2 && j >= mainW[i] + attachW[i][2]) {
                int val = dp[j - mainW[i] - attachW[i][2]] + mainV[i] + attachV[i][2];
                if (val > maxVal) maxVal = val;
            }
            
            // 选两个附件
            if (attachNum[i] >= 2 && j >= mainW[i] + attachW[i][1] + attachW[i][2]) {
                int val = dp[j - mainW[i] - attachW[i][1] - attachW[i][2]] 
                         + mainV[i] + attachV[i][1] + attachV[i][2];
                if (val > maxVal) maxVal = val;
            }
            
            if (dp[j] < maxVal) dp[j] = maxVal;
        }
    }
    printf("%d\n", dp[V]);
}
```
### 8.背包问题求方案数
```c
#define MOD 1000000007

int dp[MAXV];    // 最大价值
int cnt[MAXV];   // 方案数

void countWays(int n, int V) {
    memset(dp, 0, sizeof(dp));
    memset(cnt, 0, sizeof(cnt));
    
    // 初始化：容量为0时有一种方案
    cnt[0] = 1;
    
    for (int i = 1; i <= n; i++) {
        int weight, value;
        scanf("%d%d", &weight, &value);
        
        for (int j = V; j >= weight; j--) {
            int newVal = dp[j - weight] + value;
            if (newVal > dp[j]) {
                dp[j] = newVal;
                cnt[j] = cnt[j - weight];
            } else if (newVal == dp[j]) {
                cnt[j] = (cnt[j] + cnt[j - weight]) % MOD;
            }
        }
    }
    printf("%d %d\n", dp[V], cnt[V]);
}
```
### 9.求具体方案
```c
#include <stdio.h>
#include <string.h>

#define MAXN 100
#define MAXV 1000

int dp[MAXV];
int choose[MAXN][MAXV];
int w[MAXN];  // 添加重量数组
int v[MAXN];  // 价值数组（可选，用于调试）

void printSolution(int n, int V) {
    memset(dp, 0, sizeof(dp));
    memset(choose, 0, sizeof(choose));
    
    // 逆序输入物品
    for (int i = n; i >= 1; i--) {
        int weight, value;
        scanf("%d%d", &weight, &value);
        w[i] = weight;   // 保存重量
        v[i] = value;    // 保存价值（可选）
        
        for (int j = V; j >= weight; j--) {
            if (dp[j - weight] + value >= dp[j]) {
                dp[j] = dp[j - weight] + value;
                choose[i][j] = 1;
            }
        }
    }
    
    // 输出方案
    int vol = V;
    for (int i = 1; i <= n; i++) {
        if (choose[i][vol]) {
            printf("%d ", i);
            vol -= w[i];  // ✅ 现在 w[i] 有定义了
        }
    }
    printf("\n");
}
```
